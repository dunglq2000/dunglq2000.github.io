#!/usr/bin/env python3

from Crypto.Util.number import *
from flag import flag

def base(n, l):
    D = []
    while n > 0:
        n, r = divmod(n, l)
        D.append(r)
    return ''.join(str(d) for d in reversed(D)) or '0'

def asiv_prng(seed):
     l = len(seed)
     _seed = base(bytes_to_long(seed), 3)
     _seed = [int(_) for _ in _seed]
     _l = len(_seed)
     R = [[getRandomRange(0, 3) for _ in range(_l)] for _ in range(_l**2)]
     S = []
     for r in R:
             s = 0
             for _ in range(_l):
                     s += (r[_] * _seed[_]) % 3
             # s += getRandomRange(0, 3)
             s %= 3
             S.append(s)
     return R, S

seed = flag.lstrip(b'CCTF{').rstrip(b'}')
R, S = asiv_prng(seed)

f = open('output.txt', 'w')
f.write(f'R = {R}\nS = {S}')
f.close()

Flag được biểu diễn ở dạng cơ số :math:3, giả sử là số nguyên có dạng

\[f = s_0 + 3 s_1 + 3^2 s_2 + \cdots + 3^{l-1} s_{l-1}\]

thì _seed sẽ là vector

\[seed = (s_{l-1}, s_{l-2}, \ldots, s_1, s_0).\]

Sau đó đề sinh ma trận \((r_{ij})\) kích thước \(l^2 \times l\) gồm các phần tử ngẫu nhiên trong khoảng \([0, 3]\). Cuối cùng tính vector \(S = R \cdot seed\). Giả hệ phương trình tuyến tính ta có được flag.

#!/usr/bin/env sage

from Crypto.Util.number import *
from flag import flag

def on_barak(P, E):
     c, d, p = E
     x, y = P
     return (x**3 + y**3 + c - d*x*y) % p == 0

def add_barak(P, Q, E):
     if P == (0, 0):
             return Q
     if Q == (0, 0):
             return P
     assert on_barak(P, E) and on_barak(Q, E)
     x1, y1 = P
     x2, y2 = Q
     if P == Q:
             x3 = y1 * (c - x1**3) * inverse(x1**3 - y1**3, p) % p
             y3 = x1 * (y1**3 - c) * inverse(x1**3 - y1**3, p) % p
     else:

             x3 = (y1**2*x2 - y2**2*x1) * inverse(x2*y2 - x1*y1, p) % p
             y3 = (x1**2*y2 - x2**2*y1) * inverse(x2*y2 - x1*y1, p) % p
     return (x3, y3)

def mul_barak(m, P, E):
     if P == (0, 0):
             return P
     R = (0, 0)
     while m != 0:
             if m & 1:
                     R = add_barak(R, P, E)
             m = m >> 1
             if m != 0:
                     P = add_barak(P, P, E)
     return R

def rand_barak(E):
     c, d, p = E
     while True:
             y = randint(1, p - 1)
             K = Zmod(p)
             P.<x> = PolynomialRing(K)
             f = x**3 - d*x*y + c + y^3
             R = f.roots()
             try:
                     r = R[0][0]
                     return (r, y)
             except:
                     continue

def sox(n, d):
     x, y, t = 0, 0, d
     for s in range(n - 1):
             u = 1 & t // 2
             v = 1 & t ^ u
             x, y = spin(2**s, x, y, u, v)
             x += 2**s * u
             y += 2**s * v
             t = t // 4
     return x, y

def spin(n, x, y, u, v):
     if v == 0:
             if u == 1:
                     x = n - 1 - x
                     y = n - 1 - y
             x, y = y, x
     return x, y

def encrypt(msg, key, n):
     _msg = pad(msg, n ** 2)
     _msg, _key = [ord(_) for _ in _msg], [ord(_) for _ in key]
     img = Image.new('RGBA', (n, n), 'darkblue')

             w = len(_key)
             h = n**2 // w + 1

                             return p * q, (p, q)

def setup(msg, pkey):
     hid = sha512(msg).digest()
     while True:
             a = bytes_to_long(hid)
             if kronecker(a, pkey) == 1:
                     return a
             else:
                     hid = sha512(hid).digest()

def encrypt(msg, pkey):
     a, m = setup(msg, pkey), bytes_to_long(msg)
     B, C = bin(m)[2:], []
     for b in B:
             while True:

                     if kronecker(t, pkey) == 2 * int(b) - 1:
                             C.append((t - a * inverse(t, pkey)) % pkey)
                             break
     return (a, C)

#!/usr/bin/enc python3

import binascii
from secret import seed, flag

def gen_seed(s):
     i, j, k = 0, len(s), 0
     while i < j:
             k = k + ord(s[i])
             i += 1
     i = 0
     while i < j:
             if (i % 2) != 0:
                     k = k - (ord(s[i]) * (j - i + 1))
             else:
                     k = k + (ord(s[i]) * (j - i + 1))

             k = k % 2147483647
             i += 1

     k = (k * j) % 2147483647
     return k

def reseed(s):
     return s * 214013 + 2531011

def encrypt(s, msg):
     assert s <= 2**32
     c, d = 0, s
     enc, l = b'', len(msg)
     while c < l:
             d = reseed(d)
             enc += (msg[c] ^ ((d >> 16) & 0xff)).to_bytes(1, 'big')
             c += 1
     return enc

enc = encrypt(seed, flag)

#!/usr/bin/env sage

from Crypto.Util.number import *
from flag import flag

def gen_koymoted(nbit):
     p = getPrime(nbit)
     a, b = [randint(1, p - 1) for _ in '__']
     Ep = EllipticCurve(GF(p), [a, b])
     tp = p + 1 - Ep.order()
     _s = p ^^ ((2 ** (nbit - 1)) + 2 ** (nbit // 2))
     q = next_prime(2 * _s + 1)
     Eq = EllipticCurve(GF(q), [a, b])
     n = p * q
     tq = q + 1 - Eq.order()
     e = 65537
     while True:
             if gcd(e, (p**2 - tp**2) * (q**2 - tq**2)) == 1:
                     break
             else:
                     e = next_prime(e)
     pkey, skey = (n, e, a, b), (p, q)
     return pkey, skey

def encrypt(msg, pkey, skey):
     n, e, a, b = pkey
     p, q = skey
     m = bytes_to_long(msg)
     assert m < n
     while True:
             xp = (m**3 + a*m + b) % p
             xq = (m**3 + a*m + b) % q
             if pow(xp, (p-1)//2, p) == pow(xq, (q-1)//2, q) == 1:
                     break
             else:
                     m += 1
     eq1, eq2 = Mod(xp, p), Mod(xq, q)
     rp, rq = sqrt(eq1), sqrt(eq2)
     _, x, y = xgcd(p, q)
     Z = Zmod(n)
     x = (Z(rp) * Z(q) * Z(y) + Z(rq) * Z(p) * Z(x)) % n
     E = EllipticCurve(Z, [a, b])
     P = E(m, x)
     enc = e * P
     return enc

nbit = 256
pkey, skey = gen_koymoted(nbit)
enc = encrypt(flag, pkey, skey)

print(f'pkey = {pkey}')
print(f'enc = {enc}')

     e, l = randint(1, _o), len(msg)
     m1, m2 = bytes_to_long(msg[:l // 2]), bytes_to_long(msg[l // 2:])
     assert m1 < p and m2 < p

def sign(msg, skey):
     _tail = bytes_to_long(sha256(str(skey).encode('utf-8')).digest()) % (1 << 24)
     while True:
             K = [randint(1, 2**255) // (1 << 24) + _tail for _ in '__']

             if r * s != 0:
                     break

     t = inverse(K[0], _o) * (h * r - s * skey) % _o
     return (r, s, t)

     # u = h * r * inverse(t, _o) % _o
     # v = s * inverse(t, _o) % _o

     return _R == r

from Crypto.Util.number import *
from flag import flag

from decimal import *

def keygen(nbit, r):
     while True:
             p, q = [getPrime(nbit) for _ in '__']
             d, n = getPrime(64), p * q
             phi = (p - 1) * (q - 1)
             if GCD(d, phi) == 1:
                     e = inverse(d, phi)
                     N = bin(n)[2:-r]
                     E = bin(e)[2:-r]

                     return PKEY, pkey

def encrypt(msg, pkey, r):

Chúng ta có :math:19 vòng, mỗi vòng thứ :math:i sinh một số ngẫu nhiên \(a \in [2^{12 i}; 2^{12 i + 12}]\) và chúng ta phải gửi các số \(x, y, z\) thỏa mãn phương trình

\[x^2 + y^2 - xy = a \cdot z^3\]

lên server. Các số \(x, y, z\) không được đôi một bằng nhau hoặc không được bằng :math:0.

Ở đây, ta chọn \(x = 19a^2\), \(y = a^2\) và \(z = 7a\) thì ta có

\[x^2 + y^2 - xy = 19^2 a^4 + a^4 - 19a^4 = 343 a^4 = 7^3 a^4 = a \cdot (7a)^3.\]

Lời giải của team Black Bauhinia ở đây dùng \(x = 24a^2\), \(y = -24a^2\) và \(z = 12a\). Team bạn còn tổng quát hóa lên \(x = 3k^3a^2\), \(y = -3k^3 a^2\) và \(z = 3k^2 a\) với :math:k bất kì khác :math:0 vẫn cho lời giải đúng. Tuy nhiên lời giải của ta không có hệ số :math:3

Mấu chốt ở đây là chúng ta thấy bên trái là hàm bậc hai theo :math:x và :math:y, còn vế phải là hàm bậc ba theo :math:z. Như vậy nếu \(x = u a^2\) và \(y = v a^2\) với :math:u, :math:v là hệ số thì vế trái là hàm bậc bốn theo :math:z nên vế phải chỉ cần \(z = w a\) là xong. Tìm :math:u, :math:v và :math:w chỉ cần duyệt vét cạn một lúc là ra.
